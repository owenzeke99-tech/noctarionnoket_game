-- Local Key Caching (Hidden 24 hours)
local KEY_CACHE_NAME = "NoctarionHub_KeyCache_v6.json"
local function getSavedKey()
    if writefile and readfile and isfile then
        if isfile(KEY_CACHE_NAME) then
            local success, data = pcall(function()
                return HttpService:JSONDecode(readfile(KEY_CACHE_NAME))
            end)
            if success and data and data.key and data.time then
                if os.time() - data.time < 86400 then
                    return data.key
                end
            end
        end
    end
    return nil
end

local function saveKey(key)
    if writefile then
        pcall(function()
            local data = HttpService:JSONEncode({key = key, time = os.time()})
            writefile(KEY_CACHE_NAME, data)
        end)
    end
end

-- Button Helper Function
local function createButton(name, text, position, bgColor, textColor)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Parent = MainFrame
    btn.BackgroundColor3 = bgColor
    btn.BackgroundTransparency = 0.2
    btn.Position = position
    btn.Size = UDim2.new(0, 158, 0, 42)
    btn.Font = Enum.Font.GothamBold
    btn.Text = text
    btn.TextColor3 = textColor
    btn.TextSize = 13
    btn.AutoButtonColor = false

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 10)
    corner.Parent = btn

    -- Smooth Hover Animation
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0, Size = UDim2.new(0, 160, 0, 44), Position = position + UDim2.new(0, -1, 0, -1)}):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0.2, Size = UDim2.new(0, 158, 0, 42), Position = position}):Play()
    end)

    return btn
end

-- Smooth Shake Animation for Invalid Key
local function smoothShakeUI()
    local origin = MainFrame.Position
    local offsets = {12, -10, 8, -6, 4, -2, 0}
    
    task.spawn(function()
        for _, offset in ipairs(offsets) do
            local tween = TweenService:Create(
                MainFrame, 
                TweenInfo.new(0.045, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), 
                {Position = origin + UDim2.new(0, offset, 0, 0)}
            )
            tween:Play()
            tween.Completed:Wait()
        end
        MainFrame.Position = origin
    end)
end
